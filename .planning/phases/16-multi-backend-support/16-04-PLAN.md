**xtask `rocm_oracle.rs` skeleton** (RESEARCH §7.2 + PATTERNS Pattern F, mirroring `xtask/src/oracle_update.rs::run_oom_contract_check` and `run_helper_legacy_parity`):

```rust
//! `cargo run -p xtask -- rocm-oracle [--profile <p>]`
//!
//! Operator-driven wrapper for the ROCm full base-family oracle suite.
//! Env-gated; D-15 explicitly says no CI gate. Requires the dev host to have
//! ROCm 7.x installed and `--features rocm` available on `cintx-oracle`.

use anyhow::{Context, Result};
use std::process::Command;

pub fn run_rocm_oracle(profile: Option<&str>) -> Result<()> {
    let profile = profile.unwrap_or("base");
    // Build the feature spec. The base profile is just `rocm`. with-f12 / with-4c1e
    // get appended for those profile suffixes (mirrors `oracle_compare`'s --profiles).
    let features = if profile == "base" {
        "rocm".to_owned()
    } else {
        format!("rocm,{profile}")
    };

    let status = Command::new("cargo")
        .env("CINTX_ROCM_ORACLE", "1")
        .env("CINTX_BACKEND", "rocm")
        .args([
            "test",
            "-p",
            "cintx-oracle",
            "--features",
            &features,
            "--",
            "--ignored",
        ])
        .status()
        .context("spawn cargo test for rocm-oracle")?;

    if !status.success() {
        anyhow::bail!("rocm-oracle suite failed for profile `{profile}`");
    }
    println!("rocm-oracle suite passed for profile `{profile}`");
    Ok(())
}
```

fn parse_rocm_oracle(args: &[String]) -> Result<Command> {
    let mut profile = None;
    let mut iter = args.iter();
    while let Some(arg) = iter.next() {
        match arg.as_str() {
            "--profile" => {
                profile = Some(
                    iter.next()
                        .context("--profile requires a value")?
                        .to_owned(),
                );
            }
            other => bail!("unrecognized arg for rocm-oracle: {other}"),
        }
    }
    Ok(Command::RocmOracle { profile })
}
